#!/bin/bash
# post-startup.sh   (SAMPLE: illustrative, untested)
#
# Runs on the COS HOST as root, after payload-container is up.
# Mounts a GCS bucket with gcsfuse INSIDE the container and rsyncs
# /home/jupyter into it every 5 minutes.
#
# Use with post-startup-script-behavior=run_every_start: the container is
# recreated on every start, so the mount and the sync loop must be too.

BUCKET="my-workbench-backup"          # <-- your bucket
DEST="my-instance/home-jupyter"     # folder inside the bucket
C="payload-container"
NCTL="/var/lib/google/nerdctl/nerdctl -n k8s.io"

# 1. wait until the Jupyter container is running (about 2 minutes at most)
for i in $(seq 1 60); do
  [ "$($NCTL inspect -f '{{.State.Running}}' $C 2>/dev/null)" = "true" ] && break
  sleep 2
done

# 2. fail early, with a clear message, if the tools are not in the image
if ! $NCTL exec $C sh -c 'command -v gcsfuse >/dev/null && command -v rsync >/dev/null'; then
  echo "gcsfuse or rsync missing in $C. Alternative: gcloud storage rsync -r /home/jupyter gs://$BUCKET/$DEST" >&2
  exit 1
fi

# 3. mount the bucket at /mnt/gcs: inside the container, and NOT under
#    /home/jupyter, so rsync never copies the mount into itself
$NCTL exec $C bash -c "mkdir -p /mnt/gcs && (mountpoint -q /mnt/gcs || gcsfuse --implicit-dirs $BUCKET /mnt/gcs)"

# 4. start a detached sync loop (flock keeps it to a single copy)
$NCTL exec -d $C bash -c "
  exec 9>/tmp/gcs-sync.lock; flock -n 9 || exit 0
  while true; do
    rsync -r --inplace --size-only --no-perms --no-owner --no-group \
          --exclude='.cache/' --exclude='.container_gcloud/' \
          /home/jupyter/ /mnt/gcs/$DEST/
    sleep 300
  done
"

function drawText(g, n, lines) {
  g.querySelectorAll('text').forEach(t => t.remove());
  const big = n.h > 100;
  const t = el('text', {x: n.x + 12, y: n.y + (n.h <= 36 ? 21 : 22), class: 't'}, g); t.textContent = n.t;
  (lines || []).forEach((s, i) => { if (!s) return; const x = el('text', {x: n.x + 12, y: n.y + 40 + i * (big ? 17 : 14), class: 's'}, g); x.textContent = s; });
}

function buildScene(S) {
  Array.from(svg.childNodes).forEach(c => { if (c !== defs) svg.removeChild(c); });
  frameEls = {}; nodeEls = {}; edgeEls = {}; lblEls = {}; NODE_BY_ID = {};
  S.frames.forEach(f => {
    const g = el('g', {class: 'frame', id: 'f-' + f.id}, svg);
    el('rect', {x: f.x, y: f.y, width: f.w, height: f.h}, g);
    const t = el('text', {x: f.lx, y: f.ly + 1}, g); t.textContent = f.label;
    frameEls[f.id] = g;
  });
  S.labels.forEach(l => { const g = el('g', {class: 'lbl'}, svg); const t = el('text', {x: l.x, y: l.y}, g); t.textContent = l.text; lblEls[l.id] = g; });
  S.edges.forEach(e => {
    const g = el('g', {class: 'edge' + (e.sample ? ' sample' : ''), id: 'e-' + e.id}, svg);
    const d = e.pts.map((p, i) => (i ? 'L' : 'M') + p[0] + ',' + p[1]).join(' ');
    const path = el('path', {d, class: 'line', 'marker-end': 'url(#ar)'}, g);
    if (e.both) path.setAttribute('marker-start', 'url(#ar)');
    if (e.lbl) { const t = el('text', {x: e.l[0], y: e.l[1]}, g); t.textContent = e.lbl; }
    edgeEls[e.id] = g;
  });
  S.nodes.forEach(n => {
    NODE_BY_ID[n.id] = n;
    const g = el('g', {class: 'node k-' + n.k, id: 'n-' + n.id}, svg);
    el('rect', {x: n.x, y: n.y, width: n.w, height: n.h}, g);
    drawText(g, n, n.s);
    nodeEls[n.id] = g;
  });
}

let sc = 0, cur = 0, playing = false, timer = null, timers = [], sampleOn = $('#sample').checked;
const S_ = () => SCENES[sc];

function revealedUpTo(list, idx) {
  const nodes = new Set(), edges = new Set(), frames = new Set(), set = {};
  for (let i = 0; i <= idx; i++) {
    const s = list[i];
    s.r.n.forEach(x => nodes.add(x)); s.r.e.forEach(x => edges.add(x)); (s.r.fr || []).forEach(x => frames.add(x));
    if (s.set) for (const k in s.set) set[k] = s.set[k];
  }
  return {nodes, edges, frames, set};
}

function colourTerm(txt) {
  return txt.split('\n').map((ln, i) => {
    const esc = ln.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
    let cls = '';
    if (/^\s*\$ /.test(ln)) cls = 'c'; else if (/^\s*#/.test(ln)) cls = 'd'; else if (/(unsecured|NO PASSWORD|DENIED)/.test(ln)) cls = 'r'; else if (/(ACTIVE|Response: <Response \[200\]>)/.test(ln)) cls = 'g';
    return '<span class="ln ' + cls + '" style="animation-delay:' + Math.min(i * 45, 900) + 'ms">' + (esc || ' ') + '</span>';
  }).join('');
}

function render() {
  const S = S_(), list = active();
  if (cur >= list.length) cur = list.length - 1;
  const s = list[cur];
  timers.forEach(clearTimeout); timers = [];
  const R = revealedUpTo(list, cur);

  Object.entries(frameEls).forEach(([id, g]) => g.classList.toggle('on', R.frames.has(id)));
  Object.entries(lblEls).forEach(([id, g]) => g.classList.toggle('on', R.nodes.has(id)));
  Object.entries(nodeEls).forEach(([id, g]) => {
    g.classList.toggle('on', R.nodes.has(id));
    g.classList.toggle('focus', (s.f || []).indexOf(id) >= 0);
    const n = NODE_BY_ID[id];
    let lines = n.s;
    if (R.set[id] !== undefined && R.set[id] !== null) lines = R.set[id];
    drawText(g, n, lines);
  });
  svg.classList.toggle('hasfocus', (s.f || []).length > 0);
  Object.entries(edgeEls).forEach(([id, g]) => { g.classList.toggle('on', R.edges.has(id)); g.classList.remove('flow'); });
  (s.fl || []).forEach((id, i) => {
    const g = edgeEls[id]; if (!g) return;
    g.classList.add('on');
    if (s.seq) timers.push(setTimeout(() => g.classList.add('flow'), 500 + i * 650)); else g.classList.add('flow');
  });

  $('#caption').innerHTML = '<div class="meta"><span class="pill">' + S.tab + '</span><span class="pill">Step ' + cur + ' of ' + (list.length - 1) + '</span><span class="pill">' + s.time + '</span>' + (s.sample ? '<span class="tag smp">sample</span>' : '') + '</div>' +
    '<h3>' + s.title + '</h3><ul>' + s.what.map(w => '<li>' + w + '</li>').join('') + '</ul>' +
    (s.bar ? '<div class="bar run"><i></i></div><div class="meta">typically minutes on a real VM, shown here in a few seconds</div>' : '') +
    (s.note ? '<div class="callout' + (s.warn ? ' warn' : (s.sample ? ' smp' : '')) + '"><p>' + s.note + '</p></div>' : '');
  $('#term-title').textContent = s.evt || 'evidence';
  $('#term').innerHTML = colourTerm(ev(s.ev));

  const last = sc === SCENES.length - 1, atEnd = cur === list.length - 1;
  $('#stepno').textContent = 'Scene ' + (sc + 1) + '/' + SCENES.length + ' · step ' + cur + '/' + (list.length - 1);
  $('#b-back').disabled = sc === 0 && cur === 0;
  $('#b-next').disabled = last && atEnd;
  $('#b-next').textContent = atEnd && !last ? 'Next scene ▶' : 'Next ▶';
  $('#b-reset').disabled = cur === 0 && !playing;
  $('#b-play').textContent = playing ? '❚❚ Pause' : (last && atEnd ? '↻ Replay tour' : '▶ Play');
  document.querySelectorAll('.dot').forEach((d, i) => { d.classList.toggle('cur', i === cur); d.classList.toggle('done', i < cur); });
  document.querySelectorAll('.tab').forEach((t, i) => t.classList.toggle('cur', i === sc));
  $('#sample-wrap').hidden = S.id !== 'boot';

  const tl = $('#timeline'); tl.hidden = !S.segs;
  if (S.segs) {
    const seg = s.tl, maxSeg = list.slice(0, cur + 1).reduce((m, x) => Math.max(m, x.tl), -1);
    document.querySelectorAll('.seg').forEach((d, i) => { d.classList.toggle('cur', i === seg); d.classList.toggle('done', i < maxSeg || (i <= maxSeg && i !== seg)); });
  }
  const tn = $('#timeline-note'); if (tn) tn.hidden = !S.segs;
}

function buildDots() {
  const list = active(); const box = $('#dots'); box.innerHTML = '';

function buildTabs() {

  });

function buildTimeline(S) {
  const t = $('#timeline'); t.innerHTML = '';

function setScene(i, step) {
  sc = i; buildScene(S_()); buildTimeline(S_()); buildDots();
  const n = active().length;
  cur = step === 'last' ? n - 1 : Math.min(step, n - 1);
  render();
}

/* ---------------- controls ---------------- */
function stop() { playing = false; clearTimeout(timer); timer = null; }
function advance() {          // returns false when the whole tour is finished
  const n = active().length;
  if (cur < n - 1) { cur++; render(); return true; }
  if (sc < SCENES.length - 1) { setScene(sc + 1, 0); return true; }
  return false;
}
function tick() {
  if (!playing) return;
  if (!advance()) { stop(); render(); return; }
  const s = active()[cur];
  timer = setTimeout(tick, +$('#speed').value + (s.bar ? 1500 : 0) + (cur === 0 ? 800 : 0));
}
function play() {
  if (playing) { stop(); render(); return; }
  const last = sc === SCENES.length - 1;
  if (last && cur === active().length - 1) setScene(0, 0);
  playing = true; render();
  timer = setTimeout(tick, cur === 0 ? 1200 : +$('#speed').value);
}

  else if (sc > 0) setScene(sc - 1, 'last');

});
document.addEventListener('keydown', e => {
  if (['INPUT', 'SELECT', 'TEXTAREA'].indexOf(e.target.tagName) >= 0) return;
  const demo = $('#demo').getBoundingClientRect();
  if (demo.bottom < 0 || demo.top > innerHeight) return;
  if (e.key === ' ') { e.preventDefault(); play(); }
  else if (e.key === 'ArrowRight') { e.preventDefault(); $('#b-next').click(); }
  else if (e.key === 'ArrowLeft') { e.preventDefault(); $('#b-back').click(); }
  else if (e.key === 'Home') { e.preventDefault(); $('#b-reset').click(); }
});
