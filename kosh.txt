#!/bin/ksh
# dskit collector (A1). Thin, deterministic, resumable. No model, no network except the DataStage tools.
# Runs on the customer box (MKS ksh on the Citrix client, or ksh on the engine tier).
# Design: work/datastage-migration-accelerator/docs/06-plugin-architecture.md
#
# Rules this script keeps:
#   1. Secrets are scrubbed in the pipe. Raw tool output is never written to the snapshot.
#      Exception: DSX and XML exports are written by dsexport itself. They are scanned, not rewritten
#      (rewriting would break byte identity for the reader). Hits go to secrets.tsv as name and line only.
#   2. The password is prompted once, never stored, never logged.
#   3. Every artefact records the command that made it (without credentials), its size and hash.
#   4. Failures are listed in failures.csv and never fatal. Re-running with --resume continues.
#   5. A snapshot is never overwritten. Each run makes a new dated folder.

# 0.1.0 first build. 0.3.1: --all --reports in one run; --reports alone uses the newest snapshot. 0.3.0: --reports (design-time HTML report per job, attended). 0.2.1: password asked first, as a full line, with progress lines. 0.2.0: ./ for bare config names, --within, --selftest, --preflight, no stty on Windows, visible prompt fallback.
DSKIT_COLLECTOR_VERSION="0.3.1"

usage() {
cat <<'EOF'
Usage: dskit-collect.ksh [-c conf] ACTION [options]

Actions (pick one):
  --probe                 List assets (istool preview over PREVIEW_GLOBS). No content is extracted.
  --all                   Probe, then extract every job, sequence and server job found.
  --jobs FILE             Extract the job names listed in FILE (one per line).
  --seed NAME[,NAME...]   Extract the seed(s) and walk to their child jobs.
                          Add --within TEXT to follow only children whose category contains TEXT
                          (for example --within 0027_eComm_AdvSearch). Others are listed as "outside" in scope.txt.
  --all --reports         Extract, then the HTML reports, in one run and one snapshot (one password prompt).
  --reports               Design-time HTML report per job (dsdesign). ATTENDED: it shows a Read Only popup per job. Use
                          with --resume SNAP so it reuses the discovery. One folder per job: report.html/<job>/<job>.htm
  --selftest              Test the scrub and the shell and awk features on THIS box. No DataStage tool, no password.
  --preflight             Read-only report on this box: shell, awk, tools present. Add --tools-help to print each
                          tool's own help, and --auth to also list projects and jobs (asks for the password).

Options:
  -c FILE                 Config file (default ./dskit.conf)
  --resume DIR            Continue an existing snapshot folder instead of making a new one
  --whole-project         Also try a whole-project dsexport (routines, containers, table defs). Non-fatal.
  --stageinfo             Also run dsjob -stageinfo per stage (slow: use only for a few jobs)
  --no-history            Skip run history (-logsum)
  --dry-run               Print the plan. Run no DataStage tool.
  -h, --help              This text
EOF
}

# ---------- arguments ----------
WITHREP=0; CONF="./dskit.conf"; ACTION=""; JOBS_FILE=""; SEED=""; RESUME=""
WHOLE=0; STAGEINFO=0; HISTORY=1; DRY=0; TOOLS_HELP=0; AUTH=0; WITHIN=""
while [ $# -gt 0 ]; do
    case "$1" in
        -c) CONF="$2"; shift 2;;
        --probe) ACTION="probe"; shift;;
        --all) [ "${ACTION}" = "reports" ] && WITHREP=1; ACTION="all"; shift;;
        --jobs) ACTION="jobs"; JOBS_FILE="$2"; shift 2;;
        --seed) ACTION="seed"; SEED="$2"; shift 2;;
        --within) WITHIN="$2"; shift 2;;
        --reports) if [ -n "${ACTION}" ]; then WITHREP=1; else ACTION="reports"; fi; shift;;
        --selftest) ACTION="selftest"; shift;;
        --preflight) ACTION="preflight"; shift;;
        --tools-help) TOOLS_HELP=1; shift;;
        --auth) AUTH=1; shift;;
        --resume) RESUME="$2"; shift 2;;
        --whole-project) WHOLE=1; shift;;
        --stageinfo) STAGEINFO=1; shift;;
        --no-history) HISTORY=0; shift;;
        --dry-run) DRY=1; shift;;
        --version) echo "dskit-collect ${DSKIT_COLLECTOR_VERSION}"; exit 0;;
        -h|--help) usage; exit 0;;
        *) echo "Unknown option: $1" >&2; usage >&2; exit 2;;
    esac
done
echo "dskit-collect ${DSKIT_COLLECTOR_VERSION}"
[ -z "${ACTION}" ] && { usage >&2; exit 2; }
[ ! -f "${CONF}" ] && { echo "Config not found: ${CONF} (copy dskit.conf.example)" >&2; exit 2; }
case "${CONF}" in */*) ;; *) CONF="./${CONF}";; esac   # ksh searches PATH for a name without a slash
. "${CONF}"

# Defaults for anything the config leaves out.
if [ -z "${DSDESIGN:-}" ]; then
    case "${DSEXPORT}" in *dsexport.exe) DSDESIGN="${DSEXPORT%dsexport.exe}dsdesign.exe";; *) DSDESIGN="dsdesign";; esac
fi
: "${HISTORY_MAX:=50}"
: "${ENGINE_VERSION:=11.7}"
: "${PREVIEW_GLOBS:=Jobs/*/*.*}"
: "${SCRUB_NAME_PATTERN:=PASS|PWD|PWORD|SECRET|TOKEN|KEY|CRED|USER|LOGIN|(^|_)UN\$}"
for v in DOMAIN SERVER USER PROJECT DSJOB DSEXPORT ISTOOL OUTDIR; do
    eval "val=\${$v:-}"
    [ -z "${val}" ] && { echo "Config is missing ${v}" >&2; exit 2; }
done

umask 077

# ---------- password (asked first, once, never stored or logged) ----------
NEEDPW=1
[ "${ACTION}" = "selftest" ] && NEEDPW=0
[ "${ACTION}" = "preflight" ] && [ "${AUTH}" -eq 0 ] && NEEDPW=0
echo "Config loaded: project ${PROJECT}, user ${USER}."
if [ "${DRY}" -eq 0 ] && [ "${NEEDPW}" -eq 1 ] && [ -z "${DS_PASSWORD:-}" ]; then
    if [ -t 0 ]; then
        # The Windows console under MKS ksh cannot hide typing and stty can kill the shell there. Do not call it there.
        case "$(uname -s 2>/dev/null)" in
            Windows*|CYGWIN*|MINGW*|MSYS*) HIDE=0;;
            *) HIDE=1;;
        esac
        echo ">>> Type the password for ${USER} and press Enter."
        [ "${HIDE}" -eq 0 ] && echo ">>> (Typing is visible on this console. Run cls afterwards.)"
        [ "${HIDE}" -eq 1 ] && stty -echo 2>/dev/null
        read DS_PASSWORD
        [ "${HIDE}" -eq 1 ] && { stty echo 2>/dev/null; echo; }
        if [ -z "${DS_PASSWORD}" ]; then echo "No password entered." >&2; exit 2; fi
        echo "Password received."
    else
        echo "No DS_PASSWORD set and no terminal to prompt on." >&2; exit 2
    fi
fi
export DS_PASSWORD


# ---------- helpers ----------
now() { date '+%Y-%m-%d %H:%M:%S'; }
stamp_utc() { date -u '+%Y-%m-%dT%H:%M:%SZ'; }
json_esc() { printf '%s' "$1" | sed 's/\\/\\\\/g; s/"/\\"/g'; }

# Snapshot folder
if [ "${ACTION}" = "reports" ] && { [ -z "${RESUME}" ] || [ "${RESUME}" = "latest" ]; }; then
    RESUME="$(ls -d "${OUTDIR}"/snap-* 2>/dev/null | tail -1)"
    [ -z "${RESUME}" ] && { echo "No snapshot found under ${OUTDIR}. Run --all first." >&2; exit 2; }
    echo "Using newest snapshot: ${RESUME}"
fi
if [ -n "${RESUME}" ]; then
    SNAP="${RESUME}"
    [ ! -d "${SNAP}" ] && { echo "No such snapshot: ${SNAP}" >&2; exit 2; }
else
    SNAP="${OUTDIR}/snap-$(date '+%Y%m%d-%H%M%S')"
fi
ASSETS="${SNAP}/assets"
LOGFILE="${SNAP}/collector.log"
FAILFILE="${SNAP}/failures.csv"
SECRETS_FILE="${SNAP}/secrets.tsv"
DISCOVERED="${SNAP}/discovered.tsv"
SCOPE="${SNAP}/scope.txt"
PROGRESS="${SNAP}/_progress.txt"
echo "Creating snapshot folder ${SNAP}"
mkdir -p "${ASSETS}" "${SNAP}/probe" "${SNAP}/project"
[ ! -f "${FAILFILE}" ] && echo "Asset,Phase,Error,Timestamp" > "${FAILFILE}"
[ ! -f "${SECRETS_FILE}" ] && printf 'file\tline\tname\n' > "${SECRETS_FILE}"

log() { echo "[$(now)] $*" | tee -a "${LOGFILE}"; }
fail() { echo "\"$1\",\"$2\",\"$3\",\"$(now)\"" >> "${FAILFILE}"; log "FAIL [$1] $2: $3"; }

# Hash tool: first one that exists.
HASH_ALGO="none"
if command -v sha256sum >/dev/null 2>&1; then HASH_ALGO="sha256"; HASH_CMD="sha256sum"
elif command -v shasum >/dev/null 2>&1; then HASH_ALGO="sha256"; HASH_CMD="shasum -a 256"
elif command -v openssl >/dev/null 2>&1; then HASH_ALGO="sha256"; HASH_CMD="openssl dgst -sha256 -r"
elif command -v cksum >/dev/null 2>&1; then HASH_ALGO="cksum"; HASH_CMD="cksum"
fi
hash_file() {
    [ "${HASH_ALGO}" = "none" ] && { echo ""; return; }
    ${HASH_CMD} "$1" 2>/dev/null | awk '{print $1}'
}
size_of() { wc -c < "$1" | tr -d ' '; }


# ---------- tool wrappers (credentials live only here) ----------
dsjob_run()   { "${DSJOB}" -domain "${DOMAIN}" -server "${SERVER}" -user "${USER}" -password "${DS_PASSWORD}" "$@"; }
dsexport_run() { "${DSEXPORT}" /D="${DOMAIN}" /H="${SERVER}" /U="${USER}" /P="${DS_PASSWORD}" "$@"; }
dsdesign_run() { "${DSDESIGN}" /D="${DOMAIN}" /H="${SERVER}" /U="${USER}" /P="${DS_PASSWORD}" "$@"; }
istool_run()  { "${ISTOOL}" export -domain "${DOMAIN}" -username "${USER}" -password "${DS_PASSWORD}" "$@"; }

# ---------- scrub (stdin to stdout; names and line numbers to secrets.tsv; never values) ----------
scrub() {
    SCRUB_PAT="${SCRUB_NAME_PATTERN}" SCRUB_SEC="${SECRETS_FILE}" SCRUB_LABEL="$1" awk '
    BEGIN { pat = ENVIRON["SCRUB_PAT"]; sec = ENVIRON["SCRUB_SEC"]; lab = ENVIRON["SCRUB_LABEL"] }
    {
        line = $0; hit = ""
        # value pattern: cloud access key id (AKIA plus 16 more)
        if (match(line, /AKIA[0-9A-Z]+/) && RLENGTH >= 20) {
            line = substr(line, 1, RSTART - 1) "<SCRUBBED-KEY>" substr(line, RSTART + RLENGTH); hit = "aws-key-id"
        }
        # command-line and key=value password forms
        if (line ~ /-password[ \t]+[^ \t]+/) { gsub(/-password[ \t]+[^ \t]+/, "-password <SCRUBBED>", line); hit = "cmdline-password" }
        if (line ~ /\/P=[^ \t]+/) { gsub(/\/P=[^ \t]+/, "/P=<SCRUBBED>", line); hit = "cmdline-password" }
        if (line ~ /[Pp]assword=[^ \t;&"]+/) { gsub(/[Pp]assword=[^ \t;&"]+/, "password=<SCRUBBED>", line); hit = "password-assign" }
        # NAME = value (environment and parameter dumps)
        if (match(line, /^[ \t]*[A-Za-z0-9_.$]+[ \t]*=/)) {
            head = substr(line, 1, RLENGTH)
            name = head; sub(/[ \t]*=$/, "", name); sub(/^[ \t]*/, "", name)
            n = split(name, parts, "."); base = parts[n]; sub(/^\$/, "", base)
            if (toupper(base) ~ pat) {
                rest = substr(line, RLENGTH + 1); suffix = ""
                if (match(rest, /[ \t]+(\[[^]]*\]|\([^)]*\))([ \t]+(\[[^]]*\]|\([^)]*\)))*[ \t]*$/)) suffix = substr(rest, RSTART)
                line = head " <SCRUBBED>" suffix; hit = name
            }
        }
        print line
        if (hit != "") print lab "\t" NR "\t" hit >> sec
    }'
}

# Scan a file the tool wrote itself (DSX, XML). Report only. Never rewrites.
scan_file() {
    SCRUB_PAT="${SCRUB_NAME_PATTERN}" SCRUB_SEC="${SECRETS_FILE}" SCRUB_LABEL="$2" awk '
    BEGIN { pat = ENVIRON["SCRUB_PAT"]; sec = ENVIRON["SCRUB_SEC"]; lab = ENVIRON["SCRUB_LABEL"]; lastline = -100 }
    { sub(/\r$/, "") }
    /^[ \t]*Name[ \t]+"/ { lastname = $0; sub(/^[^"]*"/, "", lastname); sub(/".*$/, "", lastname); lastline = NR }
    {
        if (match($0, /AKIA[0-9A-Z]+/) && RLENGTH >= 20) print lab "\t" NR "\tdsx-aws-key-id" >> sec
        if ($0 ~ /-----BEGIN [A-Z ]*PRIVATE KEY/) print lab "\t" NR "\tdsx-private-key" >> sec
        if ($0 ~ /^[ \t]*(Default|DefaultValue|Value|DisplayValue)[ \t]+"[^"]+"/ && NR - lastline <= 6 && toupper(lastname) ~ pat) {
            v = $0; sub(/^[^"]*"/, "", v)
            if (v !~ /^\{iis(enc)?\}/) print lab "\t" NR "\tdsx-param:" lastname >> sec
        }
    }' "$1"
}

# Run a command, merge stderr, scrub the stream, keep the command exit code.
run_scrubbed() {
    label="$1"; out="$2"; shift 2
    rcf="${SNAP}/.rc.$$"
    ( "$@" 2>&1; echo $? > "${rcf}" ) | scrub "${label}" > "${out}"
    rc=$(cat "${rcf}" 2>/dev/null); rm -f "${rcf}"
    return "${rc:-1}"
}

# ---------- artefact bookkeeping ----------
rec() { # jobdir file tool cmdtext status
    printf '%s\t%s\t%s\t%s\n' "$2" "$3" "$4" "$5" >> "$1/.artifacts"
}

ex_dsjob() { # asset jobdir file cmdtext dsjob-args...
    asset="$1"; jd="$2"; f="$3"; cmdtxt="$4"; shift 4
    out="${jd}/${f}"
    if [ -s "${out}" ]; then rec "${jd}" "${f}" dsjob "${cmdtxt}" ok; return 0; fi
    run_scrubbed "${asset}/${f}" "${out}" dsjob_run "$@"
    rc=$?
    if [ "${rc}" -ne 0 ] || [ ! -s "${out}" ]; then
        fail "${asset}" "${f}" "rc=${rc}"; rm -f "${out}"; rec "${jd}" "${f}" dsjob "${cmdtxt}" fail; return 1
    fi
    rec "${jd}" "${f}" dsjob "${cmdtxt}" ok; return 0
}

ex_dsexport() { # asset jobdir file xmlflag
    asset="$1"; jd="$2"; f="$3"; xml="$4"
    out="${jd}/${f}"
    if [ "${xml}" = "1" ]; then cmdtxt="dsexport /XML /JOB=${asset} ${PROJECT}"; else cmdtxt="dsexport /JOB=${asset} ${PROJECT}"; fi
    if [ -s "${out}" ]; then rec "${jd}" "${f}" dsexport "${cmdtxt}" ok; return 0; fi
    errlog="${jd}/.${f}.err"
    if [ "${xml}" = "1" ]; then
        dsexport_run /XML /JOB="${asset}" "${PROJECT}" "${out}" 2>&1 | scrub "${asset}/${f}.stderr" > "${errlog}"
    else
        dsexport_run /JOB="${asset}" "${PROJECT}" "${out}" 2>&1 | scrub "${asset}/${f}.stderr" > "${errlog}"
    fi
    if [ ! -s "${out}" ]; then
        fail "${asset}" "${f}" "empty-or-error"; rm -f "${out}"; rec "${jd}" "${f}" dsexport "${cmdtxt}" fail; return 1
    fi
    scan_file "${out}" "${asset}/${f}"
    rec "${jd}" "${f}" dsexport "${cmdtxt}" ok; return 0
}

derive_params() { # from the first STARTED event of the scrubbed log (port of handover awk)
    src="$1/full.log"; out="$1/params.txt"
    [ -s "${out}" ] && { rec "$1" params.txt awk "derive from full.log" ok; return 0; }
    awk '
        /^Event Id:/ { if (capturing) exit; in_event=1; is_started=0; in_message=0; next }
        in_event && /^Type[ \t]*:[ \t]*STARTED/ { is_started=1; next }
        is_started && /^Message[ \t]*:/ { in_message=1; capturing=1; next }
        in_message && /^[^ \t]/ { in_message=0; next }
        in_message { sub(/^\t/,""); print }
    ' "${src}" > "${out}"
    if [ -s "${out}" ]; then rec "$1" params.txt awk "derive from full.log" ok
    else rm -f "${out}"; fail "$2" params.txt "empty"; rec "$1" params.txt awk "derive from full.log" fail; fi
}

derive_summary() {
    src="$1/full.log"; out="$1/summary.txt"
    [ -s "${out}" ] && { rec "$1" summary.txt awk "derive from full.log" ok; return 0; }
    awk '
        /Summary of sequence run/ { capturing=1 }
        capturing && /^Event Id:/ && started { exit }
        capturing && /^Event Id:/ { started=1 }
        capturing { print }
    ' "${src}" > "${out}"
    if [ -s "${out}" ]; then rec "$1" summary.txt awk "derive from full.log" ok; else rm -f "${out}"; rec "$1" summary.txt awk "derive from full.log" skipped; fi
}

ex_stageinfo() { # asset jobdir
    asset="$1"; jd="$2"; out="${jd}/stage_info.txt"
    [ -s "${out}" ] && { rec "${jd}" stage_info.txt dsjob "dsjob -stageinfo ${PROJECT} ${asset} <stage>" ok; return 0; }
    : > "${out}"
    while IFS= read -r stage; do
        stage=$(echo "${stage}" | tr -d '\r' | sed 's/^[[:space:]]*//;s/[[:space:]]*$//')
        [ -z "${stage}" ] && continue
        case "${stage}" in Status*) continue;; esac
        echo "### ${stage}" >> "${out}"
        dsjob_run -stageinfo "${PROJECT}" "${asset}" "${stage}" 2>/dev/null | scrub "${asset}/stage_info.txt" >> "${out}"
    done < "${jd}/stages.txt"
    rec "${jd}" stage_info.txt dsjob "dsjob -stageinfo ${PROJECT} ${asset} <stage>" ok
}

lookup_asset() { # name -> "kind<TAB>category"
    if [ -s "${DISCOVERED}" ]; then
        awk -F'\t' -v n="$1" '$1 == n { print $3 "\t" $4; found=1; exit } END { if (!found) print "unknown\t" }' "${DISCOVERED}"
    else printf 'unknown\t\n'; fi
}

write_asset_manifest() { # asset jobdir
    asset="$1"; jd="$2"; mf="${jd}/manifest.json"
    ka=$(lookup_asset "${asset}"); kind=$(echo "${ka}" | cut -f1); cat_=$(echo "${ka}" | cut -f2)
    {
        printf '{"capsule": 1,\n "asset": {"name": "%s", "kind": "%s", "category": "%s"},\n' "$(json_esc "${asset}")" "${kind}" "$(json_esc "${cat_}")"
        printf ' "source": {"platform": "datastage", "version": "%s", "project": "%s", "server": "<scrubbed>"},\n' "${ENGINE_VERSION}" "$(json_esc "${PROJECT}")"
        printf ' "extracted": {"collector": "dskit-collect %s", "stamp": "%s", "hash": "%s"},\n "artifacts": {' "${DSKIT_COLLECTOR_VERSION}" "$(stamp_utc)" "${HASH_ALGO}"
        first=1; cov=""; fails=""
        while IFS="$(printf '\t')" read -r f tool cmd st; do
            key="${f}"
            if [ "${st}" = "ok" ] && [ -f "${jd}/${f}" ]; then
                [ ${first} -eq 0 ] && printf ','
                first=0
                printf '\n  "%s": {"bytes": %s, "sha": "%s", "tool": "%s", "cmd": "%s"}' "${f}" "$(size_of "${jd}/${f}")" "$(hash_file "${jd}/${f}")" "${tool}" "$(json_esc "${cmd}")"
            fi
            cov="${cov}${cov:+, }\"${key}\": \"${st}\""
            [ "${st}" = "fail" ] && fails="${fails}${fails:+, }\"${f}\""
        done < "${jd}/.artifacts"
        printf '},\n "coverage": {%s},\n "failures": [%s]}\n' "${cov}" "${fails}"
    } > "${mf}"
}

# ---------- one asset ----------
CURRENT=""
extract_asset() {
    asset="$1"
    case "${asset}" in
        ""|*[!A-Za-z0-9_.-]*) fail "${asset}" name "unsafe-name"; return 1;;
    esac
    jd="${ASSETS}/${asset}"
    # Skip only if complete AND nothing failed. A resume therefore retries failed artefacts (existing ones are kept).
    if [ -f "${jd}/_state.txt" ] && grep -q '^COMPLETE' "${jd}/_state.txt" \
        && ! awk -F'\t' '$4=="fail" { f=1 } END { exit !f }' "${jd}/.artifacts" 2>/dev/null; then
        log "  skip (already complete): ${asset}"; return 0
    fi
    mkdir -p "${jd}"; : > "${jd}/.artifacts"; echo "START $(now)" > "${jd}/_state.txt"
    CURRENT="${asset}"

    if ex_dsjob "${asset}" "${jd}" full.log "dsjob -logdetail -full ${PROJECT} ${asset}" -logdetail -full "${PROJECT}" "${asset}"; then
        derive_params "${jd}" "${asset}"
        derive_summary "${jd}"
    fi
    [ "${HISTORY}" -eq 1 ] && ex_dsjob "${asset}" "${jd}" history.txt "dsjob -logsum -type STARTED -max ${HISTORY_MAX} ${PROJECT} ${asset}" -logsum -type STARTED -max "${HISTORY_MAX}" "${PROJECT}" "${asset}"
    ex_dsexport "${asset}" "${jd}" job.dsx 0
    ex_dsexport "${asset}" "${jd}" job.xml 1
    ex_dsjob "${asset}" "${jd}" stages.txt "dsjob -lstages ${PROJECT} ${asset}" -lstages "${PROJECT}" "${asset}" && [ "${STAGEINFO}" -eq 1 ] && ex_stageinfo "${asset}" "${jd}"
    ex_dsjob "${asset}" "${jd}" design_params.txt "dsjob -lparams ${PROJECT} ${asset}" -lparams "${PROJECT}" "${asset}"

    write_asset_manifest "${asset}" "${jd}"
    echo "COMPLETE $(now)" >> "${jd}/_state.txt"
    CURRENT=""
}

# ---------- cleanup on interrupt ----------
cleanup() {
    if [ -n "${CURRENT}" ]; then
        log "INTERRUPTED in ${CURRENT}; removing its incomplete folder"
        rm -rf "${ASSETS:?}/${CURRENT:?}"
    fi
    rm -f "${SNAP}/.rc.$$"
    log "Exit (interrupted). Continue with --resume ${SNAP}"
    exit 1
}
trap cleanup INT TERM

# ---------- probe (discovery) ----------
do_probe() {
    [ -s "${DISCOVERED}" ] && { log "Using existing discovery: ${DISCOVERED}"; return 0; }
    if [ "${DRY}" -eq 1 ]; then log "DRY: would run istool preview over: ${PREVIEW_GLOBS}"; return 0; fi
    : > "${SNAP}/probe/_all.txt"; n=0
    OLDIFS="${IFS}"; IFS=';'
    for glob in ${PREVIEW_GLOBS}; do
        IFS="${OLDIFS}"
        n=$((n + 1)); pf="${SNAP}/probe/preview_${n}.txt"
        log "istool preview: ${glob}"
        run_scrubbed "probe/preview_${n}" "${pf}" istool_run -archive "${SNAP}/probe/_discovery.isx" -preview -datastage "${SERVER}/${PROJECT}/${glob}"
        [ -s "${pf}" ] && cat "${pf}" >> "${SNAP}/probe/_all.txt"
        IFS=';'
    done
    IFS="${OLDIFS}"
    # name <TAB> ext <TAB> kind <TAB> category  (names and categories may contain spaces; split on "/" only)
    awk -v prefix="${SERVER}/${PROJECT}/" '
        /^[ \t]*\[[0-9]+\/[0-9]+\]/ {
            line = $0
            sub(/^[ \t]*\[[0-9]+\/[0-9]+\][ \t]+/, "", line); sub(/[ \t]+$/, "", line)
            p = index(line, prefix); if (p == 1) line = substr(line, length(prefix) + 1)
            n = split(line, parts, "/"); file = parts[n]
            e = match(file, /\.[A-Za-z0-9]+$/); if (!e) next
            ext = substr(file, e + 1); name = substr(file, 1, e - 1)
            cat = ""; for (i = 1; i < n; i++) cat = cat (i > 1 ? "/" : "") parts[i]
            printf "%s\t%s\t%s\t%s\n", name, ext, ext, cat
        }' "${SNAP}/probe/_all.txt" | awk -F'\t' 'BEGIN{OFS="\t"} {
            k = "other"
            if ($3=="pjb") k="parallel-job"; else if ($3=="sjb") k="server-job"; else if ($3=="qjb") k="sequence"
            else if ($3=="srt") k="routine"; else if ($3=="pst") k="parameter-set"
            print $1, $2, k, $4 }' | sort -u > "${DISCOVERED}"
    if [ ! -s "${DISCOVERED}" ]; then
        log "istool preview gave nothing. Falling back to dsjob -ljobs (jobs only, no categories)."
        dsjob_run -ljobs "${PROJECT}" 2>/dev/null | tr -d '\r' | sed 's/^[[:space:]]*//;s/[[:space:]]*$//' \
            | grep -v '^$' | grep -v '^Status' | awk '{ printf "%s\t\tother\t\n", $0 }' > "${DISCOVERED}"
    fi
    count=$(wc -l < "${DISCOVERED}" | tr -d ' ')
    log "Discovered ${count} assets. Kinds: $(cut -f3 "${DISCOVERED}" | sort | uniq -c | tr -s ' ' | tr '\n' ';')"
    [ "${count}" -eq 0 ] && { log "Empty discovery."; return 1; }
    return 0
}

# ---------- whole-project export (capture C3; allowed to fail) ----------
do_whole_project() {
    out="${SNAP}/project/whole_project.dsx"
    if [ "${DRY}" -eq 1 ]; then log "DRY: would try whole-project dsexport"; return 0; fi
    [ -s "${out}" ] && { log "Whole-project export already present."; return 0; }
    log "Trying whole-project export (routines, containers, table definitions)"
    dsexport_run "${PROJECT}" "${out}" 2>&1 | scrub "project/stderr" > "${SNAP}/project/dsexport.err"
    if [ -s "${out}" ]; then
        scan_file "${out}" "project/whole_project.dsx"
        log "Whole-project export OK: $(size_of "${out}") bytes"; echo "ok" > "${SNAP}/project/_status"
    else
        rm -f "${out}"; fail "project" whole-project-export "denied-or-empty"; echo "failed" > "${SNAP}/project/_status"
    fi
}

# ---------- scope walking ----------
children_of() { grep -i '^[[:space:]]*Jobname[[:space:]]*"' "$1" 2>/dev/null | tr -d '\r' | sed 's/^[^"]*"//; s/".*$//' | sort -u; }
routines_of() { grep -i '^[[:space:]]*Routinename[[:space:]]*"' "$1" 2>/dev/null | tr -d '\r' | sed 's/^[^"]*"//; s/".*$//' | sort -u; }

run_list() { # file of names
    total=$(grep -c . "$1" | tr -d ' '); i=0
    while IFS= read -r job; do
        job=$(echo "${job}" | tr -d '\r' | sed 's/^[[:space:]]*//;s/[[:space:]]*$//')
        [ -z "${job}" ] && continue
        i=$((i + 1)); log "[${i}/${total}] ${job}"
        printf 'TOTAL %s\nLAST_INDEX %s\nLAST_JOB %s\nHEARTBEAT %s\n' "${total}" "${i}" "${job}" "$(now)" > "${PROGRESS}"
        [ "${DRY}" -eq 1 ] && { log "DRY: would extract ${job}"; continue; }
        extract_asset "${job}"
        printf 'job\t%s\t-\n' "${job}" >> "${SCOPE}"
    done < "$1"
}

run_seed() {
    queue="${SNAP}/_queue.txt"; seen="${SNAP}/_seen.txt"; : > "${seen}"; : > "${SCOPE}.new"   # seen is per run; extract_asset skips finished assets
    echo "${SEED}" | tr ',' '\n' | sed 's/^[[:space:]]*//;s/[[:space:]]*$//' | grep -v '^$' > "${queue}"
    cp "${queue}" "${SNAP}/seeds.txt"
    i=0
    while :; do
        job=$(head -1 "${queue}"); [ -z "${job}" ] && break
        tail -n +2 "${queue}" > "${queue}.tmp"; mv "${queue}.tmp" "${queue}"
        grep -qx "${job}" "${seen}" 2>/dev/null && continue
        echo "${job}" >> "${seen}"; i=$((i + 1))
        log "[walk ${i}] ${job}"
        printf 'LAST_INDEX %s\nLAST_JOB %s\nQUEUE %s\nHEARTBEAT %s\n' "${i}" "${job}" "$(grep -c . "${queue}" | tr -d ' ')" "$(now)" > "${PROGRESS}"
        [ "${DRY}" -eq 1 ] && { log "DRY: would extract ${job} and walk its children"; continue; }
        extract_asset "${job}"
        dsx="${ASSETS}/${job}/job.dsx"
        if [ -s "${dsx}" ]; then
            for c in $(children_of "${dsx}"); do
                if [ -n "${WITHIN}" ]; then
                    cc=$(lookup_asset "${c}" | cut -f2)
                    case "${cc}" in *"${WITHIN}"*) ;; *) printf 'outside\t%s\t%s\n' "${c}" "${job}" >> "${SCOPE}.new"; continue;; esac
                fi
                printf 'job\t%s\t%s\n' "${c}" "${job}" >> "${SCOPE}.new"; echo "${c}" >> "${queue}"
            done
            for r in $(routines_of "${dsx}"); do printf 'routine-candidate\t%s\t%s\n' "${r}" "${job}" >> "${SCOPE}.new"; done
        fi
    done
    { awk '{printf "seed\t%s\t-\n", $0}' "${SNAP}/seeds.txt"; sort -u "${SCOPE}.new"; } > "${SCOPE}"; rm -f "${SCOPE}.new"
}

# ---------- run manifest ----------
write_run_manifest() {
    nasset=$(ls "${ASSETS}" 2>/dev/null | wc -l | tr -d ' ')
    nfail=$(($(grep -c . "${FAILFILE}") - 1)); nsec=$(($(grep -c . "${SECRETS_FILE}") - 1))
    {
        printf '{"capsule": 1, "run": {"collector": "dskit-collect %s", "action": "%s", "stamp": "%s", "hash": "%s", "dry_run": %s},\n' \
            "${DSKIT_COLLECTOR_VERSION}" "${ACTION}" "$(stamp_utc)" "${HASH_ALGO}" "$([ "${DRY}" -eq 1 ] && echo true || echo false)"
        printf ' "source": {"platform": "datastage", "version": "%s", "project": "%s", "server": "<scrubbed>"},\n' "${ENGINE_VERSION}" "$(json_esc "${PROJECT}")"
        printf ' "counts": {"assets": %s, "failures": %s, "secret_hits": %s},\n' "${nasset}" "${nfail}" "${nsec}"
        printf ' "whole_project": "%s", "preview_globs": "%s"}\n' "$(cat "${SNAP}/project/_status" 2>/dev/null || echo not-tried)" "$(json_esc "${PREVIEW_GLOBS}")"
    } > "${SNAP}/manifest.json"
    log "Done. assets=${nasset} failures=${nfail} secret_hits=${nsec}. Snapshot: ${SNAP}"
    [ "${nsec}" -gt 0 ] && log "NOTE: secrets.tsv lists names and line numbers only. Review before the snapshot leaves this box."
}


# ---------- design-time HTML reports (attended: dsdesign shows a Read Only popup per job) ----------
ex_report() { # asset
    asset="$1"; jd="${ASSETS}/${asset}"; inner="${jd}/report.html/${asset}/${asset}.htm"
    mkdir -p "${jd}"; [ -f "${jd}/.artifacts" ] || : > "${jd}/.artifacts"
    cmdtxt="dsdesign ${PROJECT} ${asset} /R /RP=report.html"
    if [ -s "${inner}" ]; then log "  report already present: ${asset}"; return 0; fi
    rm -rf "${jd:?}/report.html"
    dsdesign_run "${PROJECT}" "${asset}" /R /RP="${jd}/report.html" 2>&1 | scrub "${asset}/dsdesign_html.log" > "${jd}/dsdesign_html.log"
    if [ ! -s "${inner}" ]; then
        fail "${asset}" "report.html" "no-output (see dsdesign_html.log)"; rec "${jd}" "report.html/${asset}/${asset}.htm" dsdesign "${cmdtxt}" fail; return 1
    fi
    rec "${jd}" "report.html/${asset}/${asset}.htm" dsdesign "${cmdtxt}" ok
    write_asset_manifest "${asset}" "${jd}"
}
run_reports() {
    [ -s "${DISCOVERED}" ] || do_probe
    awk -F'\t' '$3=="parallel-job" || $3=="server-job" || $3=="sequence" { print $1 }' "${DISCOVERED}" > "${SNAP}/_rlist.txt"
    total=$(grep -c . "${SNAP}/_rlist.txt" | tr -d ' '); i=0
    log "ATTENDED: dsdesign shows a Read Only popup per job. Click OK each time. Jobs: ${total}"
    while IFS= read -r job; do
        [ -z "${job}" ] && continue
        i=$((i + 1)); log "[${i}/${total}] report: ${job}"
        [ "${DRY}" -eq 1 ] && { log "DRY: would run dsdesign for ${job}"; continue; }
        ex_report "${job}"
    done < "${SNAP}/_rlist.txt"
}

# ---------- selftest and preflight (read-only) ----------
selftest() {
    pass=0; bad=0
    t() { # name, input, must-contain, must-not-contain
        got=$(printf '%s\n' "$2" | scrub "selftest" 2>&1)
        okc=1
        [ -n "$3" ] && case "${got}" in *"$3"*) ;; *) okc=0;; esac
        [ -n "$4" ] && case "${got}" in *"$4"*) okc=0;; esac
        if [ ${okc} -eq 1 ]; then pass=$((pass + 1)); echo "  ok   $1"; else bad=$((bad + 1)); echo "  FAIL $1 (got: ${got})"; fi
    }
    echo "Scrub self-test (synthetic values only)"
    T=$(printf '\t')
    t "password parameter"       "${T}S3_Password=Zq-test-1"                        "<SCRUBBED>" "Zq-test-1"
    t "origin marker kept"       "${T}psDB.psPassword = Zq-test-2     (From value file)" "(From value file)" "Zq-test-2"
    t "user name value"          "${T}S3_User=svc-test-3"                           "<SCRUBBED>" "svc-test-3"
    t "key id value"             "${T}CLOUD_REF = AKIAIOSFODNN7EXAMPLE"             "<SCRUBBED-KEY>" "AKIAIOSFODNN7EXAMPLE"
    t "command line password"    "dsjob -user a -password Zq-test-4 -ljobs p"       "<SCRUBBED>" "Zq-test-4"
    t "normal value untouched"   "${T}psDB = z1p3"                                  "psDB = z1p3" ""
    t "grid setting untouched"   "${T}\$APT_GRID_ENABLE = YES   [\$PROJDEF]"        "APT_GRID_ENABLE = YES" "SCRUBBED"
    echo "Feature checks"
    [ "$(echo abc | awk '{print toupper($0)}')" = "ABC" ] && { pass=$((pass + 1)); echo "  ok   awk toupper"; } || { bad=$((bad + 1)); echo "  FAIL awk toupper"; }
    [ "$(echo abcd | awk '{print match($0,/cd$/), RSTART, RLENGTH}')" = "3 3 2" ] && { pass=$((pass + 1)); echo "  ok   awk match/RSTART/RLENGTH"; } || { bad=$((bad + 1)); echo "  FAIL awk match"; }
    [ "$(X=hi awk 'BEGIN{print ENVIRON["X"]}')" = "hi" ] && { pass=$((pass + 1)); echo "  ok   awk ENVIRON"; } || { bad=$((bad + 1)); echo "  FAIL awk ENVIRON"; }
    [ "$(echo 'a.b.c' | awk '{n=split($0,p,"."); print p[n]}')" = "c" ] && { pass=$((pass + 1)); echo "  ok   awk split on dot"; } || { bad=$((bad + 1)); echo "  FAIL awk split"; }
    [ "$(( 2 + 3 ))" = "5" ] && { pass=$((pass + 1)); echo "  ok   shell arithmetic"; } || { bad=$((bad + 1)); echo "  FAIL shell arithmetic"; }
    command -v tr >/dev/null && command -v sed >/dev/null && command -v grep >/dev/null && command -v head >/dev/null && command -v tail >/dev/null && command -v wc >/dev/null \
        && { pass=$((pass + 1)); echo "  ok   tr sed grep head tail wc present"; } || { bad=$((bad + 1)); echo "  FAIL a basic tool is missing"; }
    echo "Hash tool: ${HASH_ALGO}${HASH_CMD:+ (${HASH_CMD})}"
    echo; echo "selftest: passed ${pass}, failed ${bad}"
    [ "${bad}" -eq 0 ]
}

preflight() {
    rep="${SNAP}/preflight-report.txt"
    {
        echo "dskit preflight $(stamp_utc), collector ${DSKIT_COLLECTOR_VERSION}"
        echo "--- box"
        echo "os: $(uname -s -r -m 2>&1)"
        echo "shell: ${KSH_VERSION:-unknown} (SHELL=${SHELL:-unset})"
        echo "awk: $( (awk --version 2>&1 || awk -W version 2>&1) | head -1)"
        echo "tty on stdin: $([ -t 0 ] && echo yes || echo no)"
        echo "hash tool: ${HASH_ALGO}"
        echo "outdir writable: $(mkdir -p "${OUTDIR}" 2>/dev/null && [ -w "${OUTDIR}" ] && echo yes || echo no) (${OUTDIR})"
        echo "--- DataStage tools (from the config)"
        for tool in "${DSJOB}" "${DSEXPORT}" "${ISTOOL}"; do
            if [ -f "${tool}" ] || command -v "${tool}" >/dev/null 2>&1; then echo "present: ${tool}"; else echo "MISSING: ${tool}"; fi
        done
        for v in DSHOME DSEngine APT_ORCHHOME ISHOME; do eval "val=\${$v:-}"; [ -n "${val}" ] && echo "env ${v}=${val}"; done
        if [ "${TOOLS_HELP}" -eq 1 ]; then
            echo "--- tool help (their own text)"
            echo "## dsjob"; "${DSJOB}" -help 2>&1 | head -80
            echo "## istool"; "${ISTOOL}" -help 2>&1 | head -40
            echo "## istool export"; "${ISTOOL}" export -help 2>&1 | head -80
            echo "## dsexport"; "${DSEXPORT}" /? 2>&1 | head -60
        fi
        if [ "${AUTH}" -eq 1 ]; then
            echo "--- read-only calls with credentials"
            echo "## dsjob -lprojects"; dsjob_run -lprojects 2>&1 | head -40
            echo "## dsjob -ljobs ${PROJECT} (count only)"; echo "jobs listed: $(dsjob_run -ljobs "${PROJECT}" 2>/dev/null | grep -v '^Status' | grep -c .)"
        fi
    } 2>&1 | scrub "preflight" > "${rep}"
    cat "${rep}"
    log "Preflight report: ${rep}"
}

# ---------- main ----------
log "dskit collector ${DSKIT_COLLECTOR_VERSION}. Action=${ACTION}. Snapshot=${SNAP}"
case "${ACTION}" in
    reports) run_reports;;
    selftest) selftest; exit $?;;
    preflight) preflight; exit 0;;
    probe) do_probe;;
    all)
        do_probe || exit 1
        awk -F'\t' '$3=="parallel-job" || $3=="server-job" || $3=="sequence" { print $1 }' "${DISCOVERED}" > "${SNAP}/_list.txt"
        run_list "${SNAP}/_list.txt"
        [ "${WITHREP}" -eq 1 ] && run_reports;;
    jobs)
        [ ! -f "${JOBS_FILE}" ] && { echo "Jobs file not found: ${JOBS_FILE}" >&2; exit 2; }
        [ -s "${DISCOVERED}" ] || do_probe
        run_list "${JOBS_FILE}";;
    seed)
        [ -s "${DISCOVERED}" ] || do_probe
        run_seed;;
esac
[ "${WHOLE}" -eq 1 ] && do_whole_project
write_run_manifest
exit 0
